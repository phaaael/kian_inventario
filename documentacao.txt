Gerar uma sessão segura:

const crypto = require('crypto');

function generateSecureToken(length) {
    return crypto.randomBytes(length).toString('hex');
}

const secureSessionSecret = generateSecureToken(32);
console.log('Sessão Secreta Segura:', secureSessionSecret);
